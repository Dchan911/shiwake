        // 限定勘定科目
        const accounts = ["現金", "当座預金", "借入金", "売掛金", "買掛金", "仕入", "売上", "前払金", "前受金"];

        // State
        let currentQuestions = [...allQuestions];
        let currentIdx = 0;
        let score = 0;
        let wrongQuestionIds = [];
        let selectedSide = 'debit';
        let selection = { debit: null, credit: null };
        let isReviewMode = false;

        // DOM Elements
        let qText, qColumn, qNum, qTotal, slotDebit, slotCredit, reviewBadge;
        const accountGrid = document.getElementById('account-grid');
        const modal = document.getElementById('modal');
        const nextBtn = document.getElementById('next-btn');
        const progressBar = document.getElementById('progress');
        const scoreDisplay = document.getElementById('score');

        function initGrid() {
            accountGrid.innerHTML = '';
            accounts.forEach(acc => {
                const btn = document.createElement('button');
                btn.className = "bg-white border-2 border-gray-100 py-4 rounded-xl text-sm font-bold text-gray-700 hover:border-green-300 btn-account card shadow-sm";
                btn.textContent = acc;
                btn.onclick = () => selectAccount(acc);
                accountGrid.appendChild(btn);
            });
        }

        function loadQuestion() {
            const q = currentQuestions[currentIdx];
            if (!q) return;

            qText.textContent = q.question;
            qColumn.textContent = q.column;
            qNum.textContent = currentIdx + 1;
            qTotal.textContent = currentQuestions.length;
            selection = { debit: null, credit: null };
            updateSlots();
            selectedSide = 'debit';
            highlightActiveSlot();
            
            const progressPercent = (currentIdx / currentQuestions.length) * 100;
            progressBar.style.width = `${progressPercent}%`;

            if (isReviewMode) reviewBadge.classList.remove('hidden');
            else reviewBadge.classList.add('hidden');
        }

        function selectAccount(acc) {
            selection[selectedSide] = acc;
            updateSlots();
            if(selectedSide === 'debit' && !selection.credit) {
                selectedSide = 'credit';
            } else if (selectedSide === 'credit' && !selection.debit) {
                selectedSide = 'debit';
            }
            highlightActiveSlot();
        }

        function updateSlots() {
            slotDebit.textContent = selection.debit || "選択";
            slotCredit.textContent = selection.credit || "選択";
            slotDebit.className = `h-14 border-2 rounded-xl flex items-center justify-center font-bold transition-all ${selection.debit ? 'bg-blue-500 text-white border-blue-600' : 'border-dashed border-blue-200 text-blue-300 bg-blue-50'}`;
            slotCredit.className = `h-14 border-2 rounded-xl flex items-center justify-center font-bold transition-all ${selection.credit ? 'bg-red-500 text-white border-red-600' : 'border-dashed border-red-200 text-red-300 bg-red-50'}`;
        }

        function highlightActiveSlot() {
            slotDebit.classList.remove('ring-4', 'ring-blue-300');
            slotCredit.classList.remove('ring-4', 'ring-red-300');
            if(selectedSide === 'debit') slotDebit.classList.add('ring-4', 'ring-blue-300');
            else slotCredit.classList.add('ring-4', 'ring-red-300');
        }

        function handleAnswer() {
            if(!selection.debit || !selection.credit) return;

            const q = currentQuestions[currentIdx];
            const isCorrect = selection.debit === q.debit && selection.credit === q.credit;

            if(isCorrect) {
                if (!isReviewMode) score += 10;
                scoreDisplay.textContent = score;
                showModal("正解！✨", "ビジネススキルがアップしましたね！", "🎉");
            } else {
                if (!wrongQuestionIds.includes(q.id)) {
                    wrongQuestionIds.push(q.id);
                }
                showModal("仕訳ミス！", `正しい帳簿記録は...\n借方: 【${q.debit}】\n貸方: 【${q.credit}】`, "🤔");
            }
        }

        function showModal(title, desc, icon) {
            document.getElementById('modal-title').textContent = title;
            document.getElementById('modal-desc').innerText = desc;
            document.getElementById('modal-icon').textContent = icon;
            modal.style.display = 'flex';
        }

        nextBtn.onclick = () => {
            modal.style.display = 'none';
            currentIdx++;
            if(currentIdx < currentQuestions.length) {
                loadQuestion();
            } else {
                showFinalResult();
            }
        };

        function showFinalResult() {
            progressBar.style.width = `100%`;
            const container = document.getElementById('game-container');
            const hasWrong = wrongQuestionIds.length > 0;
            
            let html = `
                <div class="text-center py-4">
                    <div class="text-6xl mb-4">${hasWrong ? '📝' : '🏆'}</div>
                    <h2 class="text-2xl font-bold text-gray-700 mb-2">${isReviewMode ? '復習完了！' : 'ミッション完了！'}</h2>
                    <div class="text-4xl font-black text-green-500 mb-4">${score}点</div>
                    <p class="text-gray-500 mb-6 text-sm">間違えた取引: ${wrongQuestionIds.length} 件</p>
                    
                    <div class="flex flex-col gap-3">
            `;

            if (hasWrong) {
                html += `<button onclick="startReview()" class="w-full py-3 bg-orange-500 text-white rounded-xl font-bold shadow-lg active:scale-95 transition-transform">ミスした問題だけ解き直す</button>`;
            }

            html += `
                        <button onclick="resetAll()" class="w-full py-3 bg-green-500 text-white rounded-xl font-bold shadow-lg active:scale-95 transition-transform">最初から出勤する</button>
                    </div>
                </div>
            `;
            
            container.innerHTML = html;
            document.getElementById('account-area').style.display = 'none';
        }

        window.startReview = function() {
            currentQuestions = allQuestions.filter(q => wrongQuestionIds.includes(q.id));
            wrongQuestionIds = [];
            currentIdx = 0;
            isReviewMode = true;
            restoreUI();
            loadQuestion();
        }

        window.resetAll = function() {
            currentQuestions = [...allQuestions];
            wrongQuestionIds = [];
            currentIdx = 0;
            score = 0;
            scoreDisplay.textContent = "0";
            isReviewMode = false;
            restoreUI();
            loadQuestion();
        }
